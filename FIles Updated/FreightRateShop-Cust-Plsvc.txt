FUNCTION Fetch_Defaults___ (
   source_type_ IN VARCHAR2,
   source_      IN VARCHAR2) RETURN Default_Structure_Rec
IS
   rec_       Default_Structure_Rec;
   temp_weight_ NUMBER :=0;
   uom_for_weight_net_ VARCHAR2(20);
   weight_net_ NUMBER;
   
   CURSOR get_part IS 
   SELECT inventory_part_no part_no,part_catalog_api.Get_Uom_For_Weight_Net(inventory_part_no) Uom_For_Weight_Net,qty_to_ship,part_catalog_api.Get_Weight_Net(inventory_part_no) Weight_Net
   FROM Shipment_Order_Line
   WHERE shipment_order_id=source_;
   
BEGIN
   rec_.source_type_client := Frt_Source_Ref_Type_API.Decode(source_type_);
   
	rec_.ship_via_code := Frt_Shipping_Info_Util_API.Get_Ship_Via_Code(source_type_, source_);
   $IF Component_Mpccom_SYS.INSTALLED $THEN
      rec_.ship_via_desc := Mpccom_Ship_Via_API.Get_Description(rec_.ship_via_code);  
   $END
   rec_.carrier_id := Frt_Shipping_Info_Util_API.Get_Carrier_Id(source_type_, source_);
   rec_.carrier_name := Frt_Carrier_API.Get_Name(rec_.carrier_id);
   rec_.provider_id := Frt_Shipping_Info_Util_API.Get_Freight_Provider_Id(source_type_, source_);
   rec_.provider_name := Frt_Provider_Config_API.Get_Name(rec_.provider_id);
   rec_.delivery_terms := Frt_Shipping_Info_Util_API.Get_Delivery_Terms(source_type_, source_);
   $IF Component_Mpccom_SYS.INSTALLED $THEN 
      rec_.delivery_terms_desc := Order_Delivery_Term_API.Get_Description(rec_.delivery_terms);
   $END
   rec_.freight_terms_code := Frt_Shipping_Info_Util_API.Get_Freight_Terms_Code(source_type_, source_);
   
   IF (source_type_ IN ('CUSTOMER_ORDER', 'SALES_QUOTE', 'SHIPMENT_ORDER')) THEN 
      rec_.auto_pack := TRUE;
   ELSE
      rec_.auto_pack := FALSE;
   END IF;
   
   IF source_type_ = 'CUSTOMER_ORDER' THEN
      $IF Component_Order_SYS.INSTALLED $THEN 
         rec_.ship_from    := Customer_Order_API.Get_Contract(source_);
         rec_.total_weight := NVL(Customer_Order_API.Get_Total_Weight__(source_), 0);
      $ELSE
         rec_.ship_from    := NULL;
         rec_.total_weight := 0;
      $END
   ELSIF source_type_ = 'SHIPMENT' THEN
      $IF Component_Shpmnt_SYS.INSTALLED $THEN 
         rec_.ship_from    := Shipment_API.Get_Contract(source_);
         rec_.total_weight := NVL(Shipment_API.Get_Operational_Gross_Weight(source_, Company_Invent_Info_API.Get_Uom_For_Weight(Site_API.Get_Company(rec_.ship_from)), 'FALSE'), 0);
      $ELSE
         rec_.ship_from    := NULL;
         rec_.total_weight := 0;
      $END
   ELSIF source_type_ = 'SALES_QUOTE' THEN
      $IF Component_Order_SYS.INSTALLED $THEN 
         rec_.ship_from    := Order_Quotation_API.Get_Contract(source_);
         rec_.total_weight := NVL(Order_Quotation_API.Get_Total_Weight__(source_), 0);
      $ELSE
         rec_.ship_from := NULL;
         rec_.total_weight := 0;
      $END
   ELSIF source_type_ = 'SHIPMENT_ORDER' THEN
      $IF Component_Shpord_SYS.INSTALLED $THEN 
         rec_.ship_from := NULL;
         temp_weight_ :=0;      
         FOR mrec_ IN get_part LOOP
            uom_for_weight_net_ := nvl(mrec_.uom_for_weight_net,'lb');
            weight_net_ := nvl(mrec_.weight_net,1);
            IF (uom_for_weight_net_ = 'lb') THEN
               temp_weight_ := temp_weight_ + nvl(mrec_.qty_to_ship,0)*weight_net_;
            ELSIF (uom_for_weight_net_ = 'kg') THEN 
               temp_weight_ := temp_weight_ + ((nvl(mrec_.qty_to_ship,0)* weight_net_) * 2.205);
            ELSIF (uom_for_weight_net_ = 'gram') THEN 
               temp_weight_ := temp_weight_ + ((nvl(mrec_.qty_to_ship,0)* weight_net_) * 0.00220462);
            ELSIF (uom_for_weight_net_ = 't') THEN
               temp_weight_ := temp_weight_ + ((nvl(mrec_.qty_to_ship,0)* weight_net_) * 2000);
            ELSE
               temp_weight_ := temp_weight_ + nvl(mrec_.qty_to_ship,0)*weight_net_;
            END IF;
         END LOOP;
         rec_.total_weight := temp_weight_;
         rec_.sender_type := Shipment_Order_API.Get_Sender_Type_Db(source_);
         rec_.sender_id := Shipment_Order_API.Get_Sender_Id(source_);
      $ELSE
         rec_.ship_from := NULL;
         rec_.total_weight := 0;
      $END
   ELSE
      rec_.ship_from := NULL;
      rec_.total_weight := 0;
   END IF;
   
   rec_.weight_uom := Company_Invent_Info_API.Get_Uom_For_Weight(Site_API.Get_Company(rec_.ship_from));
   
   IF (source_type_ IN ('CUSTOMER_ORDER', 'SALES_QUOTE', 'SHIPMENT')) THEN 
      rec_.sender_type := Sender_Receiver_Type_API.DB_SITE;
      rec_.sender_id := rec_.ship_from;
   END IF;
   
   RETURN rec_; 
END Fetch_Defaults___;


@DynamicComponentDependency DISCOM  
FUNCTION Fetch_Shipping_Rates___ (
   transaction_id_ IN VARCHAR2,
   source_ref_     IN VARCHAR2,
   source_         IN VARCHAR2,
   auto_pack_      IN BOOLEAN,
   ship_via_code_  IN VARCHAR2,
   frt_provider_   IN VARCHAR2,
   sender_type_    IN VARCHAR2,
   sender_id_      IN VARCHAR2,
   objkey_         IN VARCHAR2) RETURN Rate_Out_Structure_Rec
IS
   shipment_rates_out_  Frt_Int_Message_Util_API.Sr_Shipment_Rates_Rec;
   error_message_       VARCHAR2(2000);
   origin_              Frt_Int_Message_Util_API.Sp_Origin_Rec := NULL;
   rec_                 Rate_Out_Structure_Rec;
   line_rec_            Rate_Details_Virtual_Rec;
   parent_key_          Freight_Rate_Shop_Virtual_Key;
   parent_rec_          Freight_Rate_Shop_Virtual_Rec;
   fastest_             BOOLEAN;
   lowest_cost_         BOOLEAN;
   auto_pack_str_       VARCHAR2(5) := CASE auto_pack_
                                          WHEN TRUE THEN 'TRUE'
                                          WHEN FALSE THEN 'FALSE'
                                          ELSE 'NULL'
                                       END;
BEGIN
                                       
   origin_ := Get_Origin_Rec___(frt_provider_, sender_type_, sender_id_);
   
   IF (origin_.location_type IS NULL OR
       origin_.location_site IS NULL OR
       origin_.location_code IS NULL) THEN
      Error_SYS.Record_General(lu_name_, 'NOFRTPROVIDERLOCATION: No freight provider location defined for this site/remote warehouse');
   END IF;
                                
   IF source_ref_ = 'CUSTOMER_ORDER' THEN
      Frt_Int_Ship_Util_API.Get_Cust_Order_Rates(shipment_rates_out_, transaction_id_, source_, error_message_, origin_, auto_pack_str_, ship_via_code_, frt_provider_);
   ELSIF source_ref_ = 'SHIPMENT' THEN
      Frt_Int_Ship_Util_API.Get_Shipment_Rates(shipment_rates_out_, transaction_id_, source_, error_message_, origin_, auto_pack_str_, ship_via_code_, frt_provider_);
   ELSIF source_ref_ = 'SALES_QUOTE' THEN 
      Frt_Int_Ship_Util_API.Get_Quotation_Rates(shipment_rates_out_, transaction_id_, source_, error_message_, origin_, auto_pack_str_, ship_via_code_, frt_provider_);
   ELSIF source_ref_ = 'SHIPMENT_ORDER' THEN 
      Frt_Int_Ship_Util_API.Get_Shipment_Order_Rates(shipment_rates_out_, transaction_id_, source_, error_message_, origin_, auto_pack_str_, ship_via_code_, frt_provider_);
   END IF;
   
      
   rec_.lowest_cost_carrier            :=  shipment_rates_out_.service_recommendation.lowest_cost_carrier_scac;
   rec_.lowest_cost_carrier_name       :=  shipment_rates_out_.service_recommendation.lowest_cost_carrier_number;
   rec_.lowest_cost_service            :=  shipment_rates_out_.service_recommendation.lowest_cost_carrier_service_code_description;
   rec_.lowest_cost_arrival_date       :=  shipment_rates_out_.service_recommendation.lowest_cost_arrival_date_text;
   rec_.lowest_cost_currency           :=  shipment_rates_out_.service_recommendation.lowest_cost_currency_code;
   rec_.lowest_cost_ship_via           :=  shipment_rates_out_.service_recommendation.lowest_cost_ship_code_x_ref;
   rec_.lowest_cost_total_cost         :=  shipment_rates_out_.service_recommendation.lowest_cost_consignor_freight;
   rec_.lowest_cost_total_price        :=  shipment_rates_out_.service_recommendation.lowest_cost_consignee_freight;
   rec_.lowest_cost_total_list_price   :=  shipment_rates_out_.service_recommendation.lowest_cost_list_freight;
   rec_.fastest_carrier                :=  shipment_rates_out_.service_recommendation.fastest_carrier_scac;
   rec_.fastest_carrier_name           :=  shipment_rates_out_.service_recommendation.fastest_carrier_number;
   rec_.fastest_service                :=  shipment_rates_out_.service_recommendation.fastest_carrier_service_code_description;
   rec_.fastest_arrival_date           :=  shipment_rates_out_.service_recommendation.fastest_arrival_date_text;
   rec_.fastest_currency               :=  shipment_rates_out_.service_recommendation.fastest_currency_code;
   rec_.fastest_ship_via               :=  shipment_rates_out_.service_recommendation.fastest_ship_code_x_ref;
   rec_.fastest_total_cost             :=  shipment_rates_out_.service_recommendation.fastest_consignor_freight;
   rec_.fastest_total_price            :=  shipment_rates_out_.service_recommendation.fastest_consignee_freight;
   rec_.fastest_total_list_price       :=  shipment_rates_out_.service_recommendation.fastest_list_freight;
   parent_key_.objkey                  := objkey_;
   parent_rec_                         := Get_Record___(parent_key_);
   Delete_Rate_Details___(parent_key_);
   
   
   IF shipment_rates_out_.rating_list IS NOT NULL THEN
      FOR i IN shipment_rates_out_.rating_list.FIRST..shipment_rates_out_.rating_list.LAST LOOP
         line_rec_.objkey              := sys_guid();
         line_rec_.parent_objkey       := parent_rec_.objkey;
         line_rec_.name                := shipment_rates_out_.rating_list(i).carrier_number;
         line_rec_.service             := shipment_rates_out_.rating_list(i).carrier_service_code_description;
         line_rec_.ship_via            := shipment_rates_out_.rating_list(i).ship_code_x_ref;
         line_rec_.total_cost          := shipment_rates_out_.rating_list(i).consignor_freight;
         line_rec_.total_price         := shipment_rates_out_.rating_list(i).consignee_freight;
         line_rec_.total_list_price    := shipment_rates_out_.rating_list(i).list_freight;
         line_rec_.currency            := shipment_rates_out_.rating_list(i).currency_code;
         line_rec_.arrival_date        := shipment_rates_out_.rating_list(i).arrival_date_text;
         line_rec_.select_rate         := FALSE;
         
         lowest_cost_ := (line_rec_.name = rec_.lowest_cost_carrier_name AND line_rec_.ship_via = rec_.lowest_cost_ship_via);
         fastest_     := (line_rec_.name = rec_.fastest_carrier_name AND line_rec_.ship_via = rec_.fastest_ship_via);  
         
         IF (lowest_cost_ AND fastest_) THEN
            line_rec_.rate_flag := 'Lowest Cost + Fastest';
         ELSIF (lowest_cost_) THEN
            line_rec_.rate_flag := 'Lowest Cost';
         ELSIF (fastest_) THEN
            line_rec_.rate_flag := 'Fastest';
         ELSE
            line_rec_.rate_flag := '';
         END IF;
         
         
         CRUD_Create___(line_rec_);
      END LOOP;
   END IF;
   
   RETURN rec_;
END Fetch_Shipping_Rates___;


PROCEDURE Delete_Rate_Details___ (
   parent_key_ IN Freight_Rate_Shop_Virtual_Key)
IS
   line_key_ Rate_Details_Virtual_Key;
   line_rec_ Rate_Details_Virtual_Rec;
BEGIN
   FOR rec_ IN Get_Rate_Details_Array___(parent_key_) LOOP
      line_key_.objkey := rec_.objkey;
      line_rec_        := Get_Record___(line_key_);
      CRUD_Delete___(line_rec_);
   END LOOP;
END Delete_Rate_Details___;

PROCEDURE Create_Frt_Rate_Transaction___ (
   source_ref_    IN VARCHAR2,
   source_        IN VARCHAR2,
   parent_objkey_ IN VARCHAR2)
IS
   parent_key_          Freight_Rate_Shop_Virtual_Key; 
   parent_rec_          Freight_Rate_Shop_Virtual_Rec;
   line_key_            Rate_Details_Virtual_Key;
   line_rec_            Rate_Details_Virtual_Rec;
   rate_selected_       BOOLEAN := FALSE;
BEGIN
	parent_key_.objkey                  := parent_objkey_;
   parent_rec_                         := Get_Record___(parent_key_);
   
   FOR rec_ IN Get_Rate_Details_Array___(parent_key_) LOOP
      line_key_.objkey           := rec_.objkey;
      line_rec_                  := Get_Record___(line_key_);
      
      IF (line_rec_.select_rate) THEN
         Frt_Transaction_Util_API.Create_Rate_Shop(source_ref_, 
                                                   source_, 
                                                   'RATE', 
                                                   line_rec_.name, 
                                                   line_rec_.service, 
                                                   line_rec_.ship_via, 
                                                   line_rec_.total_cost, 
                                                   line_rec_.total_price, 
                                                   line_rec_.total_list_price, 
                                                   parent_rec_.total_weight, 
                                                   line_rec_.currency, 
                                                   line_rec_.arrival_date);
         rate_selected_ := TRUE;
         EXIT;
      END IF;
      
   END LOOP;
   
   IF (NOT rate_selected_) THEN
      Error_SYS.Record_General(lu_name_, 'NORATESELECTED: No rate selected.');
   END IF;
   
   
END Create_Frt_Rate_Transaction___;


FUNCTION Get_Origin_Rec___ (
   freight_provider_id_ IN VARCHAR2,
   sender_type_         IN VARCHAR2,
   sender_id_           IN VARCHAR2) RETURN Frt_Int_Message_Util_API.Sp_Origin_Rec
IS
   CURSOR get_attributes IS
      SELECT   'Facility'              location_type,
               freight_site_code       location_site,
               freight_facility_code   location_code
      FROM frt_provider_location_tab
      WHERE freight_provider_id = freight_provider_id_
      AND   sender_type = sender_type_
      AND   sender_id = sender_id_;
         
   attribute_rec_    get_attributes%ROWTYPE;   
   return_rec_       Frt_Int_Message_Util_API.Sp_Origin_Rec;   
BEGIN
   OPEN get_attributes;
   FETCH get_attributes INTO attribute_rec_;
   return_rec_.location_type := attribute_rec_.location_type;
   return_rec_.location_site := attribute_rec_.location_site;
   return_rec_.location_code := attribute_rec_.location_code;
   CLOSE get_attributes;
   
   RETURN return_rec_;
END Get_Origin_Rec___;


PROCEDURE Check_Delay___ (
   wait_delay_ IN NUMBER)
IS
   stime_ NUMBER;
   etime_ NUMBER;
   cnt_  NUMBER;
   tme_  NUMBER;
   loop_time_ NUMBER;
BEGIN
	stime_ := to_number(to_char(sysdate,'SS'));
   --dbms_output.put_line(stime||'  '||etime);
   loop_time_ := wait_delay_;
   IF (nvl(wait_delay_,0)=0 ) THEN
      loop_time_ := 5;
   END IF;
   cnt_ := 0;
   WHILE cnt_ < 1 LOOP 
      etime_ := to_number(to_char(sysdate,'SS'));
      tme_ := abs(etime_ - stime_);
      IF tme_ > wait_delay_ THEN
         cnt_ := 2;
      END IF;
   END LOOP;
 --  dbms_output.put_line(stime||'  '||etime);
   IF (etime_ > stime_ ) THEN
     -- dbms_output.put_line('TRUE');
      NULL;
   END IF;
END Check_Delay___;
