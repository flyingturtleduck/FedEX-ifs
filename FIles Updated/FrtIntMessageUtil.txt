FUNCTION Check_Is_Null___ (
   rec_ IN Sp_Shipment_Option_Arr) RETURN BOOLEAN
IS
   is_null_   BOOLEAN := TRUE;
BEGIN
   IF (rec_.count = 0) THEN
      RETURN TRUE;
   END IF;
   FOR i IN rec_.FIRST..rec_.LAST LOOP
      IF (rec_(i).value IS NOT NULL) THEN
         RETURN FALSE;
      END IF;
   END LOOP;
   RETURN TRUE;   
END Check_Is_Null___;

PROCEDURE Get_Shipment_Options_Rec(
   source_ref_       IN  VARCHAR2,
   source_ref_type_  IN  VARCHAR2,
   shipment_options_ OUT Sp_Shipment_Option_Arr,
   ltl_options_      OUT Sp_Shipment_Option_Arr)
IS    
   value_             VARCHAR2(200);
   
   CURSOR get_incldued_services IS
         SELECT ss.service_code, ss.service_value, bs.tag_name, bs.service_type
         FROM frt_ship_service_tab bs, frt_shipping_service_tab ss
         WHERE bs.service_code  = ss.service_code
         AND ss.source_ref_type = source_ref_type_
         AND ss.source_ref = source_ref_
         AND ss.include_service = 'TRUE';   
BEGIN
   shipment_options_ := Sp_Shipment_Option_Arr();
   ltl_options_      := Sp_Shipment_Option_Arr();
   
   FOR rec_ IN get_incldued_services LOOP 
      IF (rec_.service_type = Frt_Service_Type_API.DB_SHIPMENT_OPTION) THEN 
         shipment_options_.extend;
         shipment_options_(shipment_options_.LAST).name := rec_.tag_name;
         shipment_options_(shipment_options_.LAST).value := rec_.service_value;
      ELSIF (rec_.service_type = Frt_Service_Type_API.DB_LTL_OPTION) THEN 
         ltl_options_.extend;
         ltl_options_(ltl_options_.LAST).name := rec_.tag_name;
         ltl_options_(ltl_options_.LAST).value := rec_.service_value;
      END IF;
   END LOOP;
END Get_Shipment_Options_Rec;

FUNCTION Get_Ship_Service_Codes_Arr(
   source_ref_type_  IN  VARCHAR2,
   source_ref_       IN  VARCHAR2 ) RETURN Sp_Shipment_Service_Codes_Arr
IS    
   return_arr_              Sp_Shipment_Service_Codes_Arr := Sp_Shipment_Service_Codes_Arr();
   
   -- Shipment services that are codes in separate Shipment_Rec array, not Shipment or LTL options
   CURSOR get_ship_svc_codes IS
         SELECT ss.service_code, ss.service_value, bs.tag_name, bs.service_type
         FROM frt_ship_service_tab bs, frt_shipping_service_tab ss
         WHERE bs.service_code  = ss.service_code
         AND ss.source_ref_type = source_ref_type_
         AND ss.source_ref = source_ref_
         AND ss.include_service = 'TRUE';
BEGIN
   
   FOR rec_ IN get_ship_svc_codes LOOP 
      IF (rec_.service_type = Frt_Service_Type_API.DB_SHIPMENT_SERVICES_CODE) THEN 
         return_arr_.extend;
         return_arr_(return_arr_.last).code := rec_.tag_name;
      END IF;
   END LOOP;
   
   RETURN return_arr_;
END Get_Ship_Service_Codes_Arr;


FUNCTION Shipment_Rec_To_Json (
   rec_         IN Sp_Shipment_Rec,
   add_headers_ IN BOOLEAN DEFAULT FALSE,
   use_address_ IN BOOLEAN DEFAULT FALSE) RETURN Json_Object_T
IS
   
   json_top_    Json_Object_T := Json_Object_T();
   json_        Json_Object_T := Json_Object_T();
   json_array_  Json_Array_T;
   
BEGIN
   json_.put('Location', rec_.location);
   json_.put('ExternalTransactionId', rec_.external_transaction_id);
   json_.put('TransactionId', rec_.transaction_id);
   
   -- Add additional mapped tags
   IF (rec_.shipment_map_tags IS NOT NULL) THEN
      IF (rec_.shipment_map_tags.count > 0) THEN
         FOR i IN rec_.shipment_map_tags.first .. rec_.shipment_map_tags.last LOOP
            IF(rec_.shipment_map_tags(i).name IS NOT NULL AND rec_.shipment_map_tags(i).value IS NOT NULL) THEN
               json_.put(rec_.shipment_map_tags(i).name, rec_.shipment_map_tags(i).value);
            END IF;
         END LOOP;
      END IF;
   END IF;
   
   IF (NOT Check_Is_Null___(rec_.origin)) THEN
      json_.Put('Origin', Origin_Rec_To_Json(rec_.origin, FALSE, use_address_));
   ELSE
      json_.Put_Null('Origin');
   END IF;
   IF (NOT Check_Is_Null___(rec_.destination)) THEN
      json_.Put('Destination', Destination_Rec_To_Json(rec_.destination, FALSE));
   ELSE
      json_.Put_Null('Destination');
   END IF;
   IF (NOT Check_Is_Null___(rec_.carrier_details)) THEN
      json_.Put('CarrierDetails', Carrier_Details_Rec_To_Json(rec_.carrier_details, FALSE));
   ELSE
      json_.Put_Null('CarrierDetails');
   END IF;
   IF (NOT Check_Is_Null___(rec_.billing_details)) THEN
      json_.Put('BillingDetails', Billing_Details_Rec_To_Json(rec_.billing_details, FALSE));
   ELSE
      json_.Put_Null('BillingDetails');
   END IF;
   IF (NOT Check_Is_Null___(rec_.shipment_detail)) THEN
      json_.Put('ShipmentDetail', Shipment_Detail_Rec_To_Json(rec_.shipment_detail, FALSE));
   ELSE
      json_.Put_Null('ShipmentDetail');
   END IF;
   
   -- (+) 20241229 jomuus CS0206131 FI Shipping Services Mod (START)
   json_array_ := Json_Array_T();
   IF (rec_.shipment_service_codes IS NOT NULL) THEN
      IF (rec_.shipment_service_codes.count > 0) THEN
         json_array_ := Ship_Service_Codes_Arr_To_Json(rec_.shipment_service_codes);
      END IF;
   END IF;
   json_.Put('shipmentServicesCodes', json_array_);
   -- (+) 20241229 jomuus CS0206131 FI Shipping Services Mod (FINISH)
   
   json_array_ := Json_Array_T();
   IF (rec_.message_list IS NOT NULL) THEN
      IF (rec_.message_list.count > 0) THEN
         json_array_ := Shipment_Message_Arr_To_Json(rec_.message_list);
      END IF;
   END IF;
   json_.Put('MessageList', json_array_);
   
   json_array_ := Json_Array_T();
   IF (rec_.custom_fields IS NOT NULL) THEN
      IF (rec_.custom_fields.count > 0) THEN
         json_array_ := Shipment_Cf_Arr_To_Json(rec_.custom_fields);
      END IF;
   END IF;
   json_.Put('CustomFields', json_array_);
   
   json_array_ := Json_Array_T();

FUNCTION Product_Details_Rec_To_Json (
   rec_         IN Sp_Product_Details_Rec,
   add_headers_ IN BOOLEAN DEFAULT FALSE) RETURN Json_Object_T
IS  
   json_top_    Json_Object_T := Json_Object_T();
   json_        Json_Object_T := Json_Object_T();
   json_array_  Json_Array_T;
   
BEGIN
   json_.put('Number', rec_.number);
   json_.put('Description', rec_.description);
   json_.put('CommodityName', rec_.commodity_name);
   
   Json_Put___(json_, 'Weight', rec_.weight);
   Json_Put___(json_, 'AutoPack', rec_.auto_pack);
   --Json_Put___(json_, 'packUIRmngItem', rec_.pack_ui_rmng_item);
   
   IF (NOT Check_Is_Null___(rec_.dimensions)) THEN
      json_.Put('Dimensions', Dimensions_Rec_To_Json(rec_.dimensions, FALSE));
   ELSE
      json_.Put_Null('Dimensions');
   END IF;
   
   IF (NOT Check_Is_Null___(rec_.quantity)) THEN
      json_.Put('Quantity', Quantity_Rec_To_Json(rec_.quantity, FALSE));
   ELSE
      json_.Put_Null('Quantity');
   END IF;
   IF (NOT Check_Is_Null___(rec_.cost)) THEN
      json_.Put('Cost', Cost_Rec_To_Json(rec_.cost, FALSE));
   ELSE
      json_.Put_Null('Cost');
   END IF;
   IF (NOT Check_Is_Null___(rec_.price)) THEN
      json_.Put('Price', Price_Rec_To_Json(rec_.price, FALSE));
   ELSE
      json_.Put_Null('Price');
   END IF;
   
   -- Add additional mapped tags
   IF (rec_.product_detail_map_tags IS NOT NULL) THEN
      IF (rec_.product_detail_map_tags.count > 0) THEN
         FOR i IN rec_.product_detail_map_tags.first .. rec_.product_detail_map_tags.last LOOP
            IF(rec_.product_detail_map_tags(i).name IS NOT NULL AND rec_.product_detail_map_tags(i).value IS NOT NULL) THEN
               json_.put(rec_.product_detail_map_tags(i).name, rec_.product_detail_map_tags(i).value);
            END IF;
         END LOOP;
      END IF;
   END IF;
   
   json_array_ := Json_Array_T();
   IF (rec_.custom_fields IS NOT NULL) THEN
      IF (rec_.custom_fields.count > 0) THEN
         json_array_ := Shipment_Cf_Arr_To_Json(rec_.custom_fields);
      END IF;
   END IF;
   json_.Put('CustomFields', json_array_);
   
   IF (add_headers_) THEN
      json_top_.Put('ProductDetails', json_);
      RETURN json_top_;
   END IF;
   RETURN json_;                
END Product_Details_Rec_To_Json;

FUNCTION Get_Package_Details_Arr (
   freight_provider_ IN VARCHAR2,
   shipment_id_      IN NUMBER) RETURN Sp_Package_Details_Arr
IS
   return_arr_           Sp_Package_Details_Arr := Sp_Package_Details_Arr();
   product_details_      Sp_Product_Details_Arr := Sp_Product_Details_Arr();
   price_rounding_       NUMBER;
   cost_rounding_        NUMBER;
   company_              VARCHAR2(30);
   contract_             VARCHAR2(5);
   total_weight_         NUMBER;
   attr_                 VARCHAR2(500);
   
   CURSOR top_level_handl_units IS
         SELECT handling_unit_type_id, handling_unit_id, operative_gross_weight weight, upper(uom_for_weight) units
         FROM handling_unit_shipment
         WHERE shipment_id = shipment_id_
         AND structure_level = 1;
   
   CURSOR get_unattached_items IS
         SELECT source_part_no, source_part_description, source_unit_meas, remaining_qty_to_attach quantity,
            source_ref1, source_ref2, source_ref3, source_ref4,
            --Customer_Order_Line_API.Get_Part_Cost(contract_, source_part_no) unit_cost,
            Inventory_Part_Cost_API.Get_Part_Cost(contract_, Sales_Part_API.Get_Part_No(contract_, source_part_no))*conv_factor unit_cost,
            Customer_Order_Line_API.Get_Sale_Unit_Price(source_ref1, source_ref2, source_ref3, source_ref4) unit_price,
            Customer_Order_API.Get_Currency_Code(source_ref1) sales_currency_code,
            Company_Finance_API.Get_Currency_Code(company_) cost_currency_code,
            Inventory_Part_API.Get_Prime_Commodity(contract_, Sales_Part_API.Get_Part_No(contract_, source_part_no)) commodity,
            Sales_Part_API.Get_Gross_Weight(contract_, source_part_no) unit_weight
         FROM shipment_connectable_line
         WHERE shipment_id = shipment_id_ 
         AND remaining_qty_to_attach > 0;
BEGIN
   -- add attached shipment lines (packed items):
   FOR rec_ IN top_level_handl_units LOOP
      return_arr_.extend;
      IF (rec_.weight > 0) THEN 
         return_arr_(return_arr_.last).weight := round(rec_.weight, 4);
      END IF;
      IF (NVL(Frt_Provider_Config_Value_API.Get_Config_Value(freight_provider_, 'SEND_CONTAINER_NO'), 'FALSE') = 'TRUE') THEN
         return_arr_(return_arr_.last).package_number := rec_.handling_unit_type_id;
      END IF;
      return_arr_(return_arr_.last).dimensions := Get_Dimensions_Rec(shipment_id_, rec_.handling_unit_id);
      return_arr_(return_arr_.last).product_details_list := Get_Product_Details_Arr(freight_provider_, shipment_id_, rec_.handling_unit_id);
      
      Client_SYS.Clear_Attr(attr_);
      Client_SYS.Add_To_Attr('HANDLING_UNIT_ID', rec_.handling_unit_id, attr_);
      return_arr_(return_arr_.last).custom_fields := Get_Custom_Fields_Arr(freight_provider_, Frt_Message_Level_API.DB_HANDLING_UNIT, Frt_Source_Ref_Type_API.DB_SHIPMENT, to_char(shipment_id_), attr_);
      Get_Package_User_Fields(freight_provider_, return_arr_(return_arr_.last), Frt_Message_Level_API.DB_HANDLING_UNIT, Frt_Source_Ref_Type_API.DB_SHIPMENT, to_char(shipment_id_), attr_);
   END LOOP;
   
   -- add unattcahed items (unpacked items)
   contract_ := Shipment_API.Get_Contract(shipment_id_);
   company_ := Site_API.Get_Company(contract_);
   cost_rounding_ := Nvl(Currency_Code_API.Get_Currency_Rounding(company_, Company_Finance_API.Get_Currency_Code(company_)), 2);
   price_rounding_ := Nvl(Currency_Code_API.Get_Currency_Rounding(company_, Customer_Order_API.Get_Currency_Code(Shipment_API.Get_Source_Ref1(shipment_id_))), 2);  
   
   total_weight_ := 0;
   FOR rec_ IN get_unattached_items LOOP  
      product_details_.extend;
      product_details_(product_details_.last).number := rec_.source_part_no;
      product_details_(product_details_.last).description := rec_.source_part_description;
      product_details_(product_details_.last).commodity_name := rec_.commodity;
      --product_details_(product_details_.last).pack_ui_rmng_item := 'Y';
      product_details_(product_details_.last).quantity.units := rec_.source_unit_meas;
      product_details_(product_details_.last).quantity.value := rec_.quantity;
      product_details_(product_details_.last).cost.currency := rec_.cost_currency_code ;
      product_details_(product_details_.last).cost.amount := round(rec_.unit_cost, cost_rounding_);
      product_details_(product_details_.last).price.currency := rec_.sales_currency_code;
      product_details_(product_details_.last).price.amount := round(rec_.unit_price, price_rounding_); 
      product_details_(product_details_.last).auto_pack := 'true';
      
      Client_SYS.Clear_Attr(attr_);
      Client_SYS.Add_To_Attr('ORDER_NO', rec_.source_ref1, attr_);
      Client_SYS.Add_To_Attr('LINE_NO', rec_.source_ref2, attr_);
      Client_SYS.Add_To_Attr('REL_NO', rec_.source_ref3, attr_);
      Client_SYS.Add_To_Attr('LINE_ITEM_NO', rec_.source_ref4, attr_);
      Client_SYS.Add_To_Attr('CONTRACT', contract_, attr_);
      Client_SYS.Add_To_Attr('CATALOG_NO', rec_.source_part_no, attr_);
      Client_SYS.Add_To_Attr('PART_NO', Sales_Part_API.Get_Part_No(contract_, rec_.source_part_no), attr_);
      --(+) 240327 DIKUUS START Add get map tags
      product_details_(product_details_.last).product_detail_map_tags := Get_Map_Tags_Arr(freight_provider_,
                                                                                Frt_Message_Level_API.DB_LINE_ITEM,
                                                                                Frt_Map_Tag_Type_API.DB_PRODUCT_DETAIL_TAG,
                                                                                Frt_Source_Ref_Type_API.DB_SHIPMENT,
                                                                                to_char(shipment_id_),
                                                                                attr_);
      --(+) 240327 DIKUUS FINISH
      product_details_(product_details_.last).custom_fields := Get_Custom_Fields_Arr(freight_provider_, Frt_Message_Level_API.DB_LINE_ITEM, Frt_Source_Ref_Type_API.DB_SHIPMENT, to_char(shipment_id_), attr_);    
      product_details_(product_details_.last).dimensions := Get_Product_Dimensions_Rec(freight_provider_, Frt_Message_Level_API.DB_LINE_ITEM, Frt_Source_Ref_Type_API.DB_SHIPMENT, to_char(shipment_id_), attr_);
      total_weight_ := total_weight_ + (rec_.quantity * nvl(rec_.unit_weight, 0));
   END LOOP;
   
   IF (product_details_.COUNT > 0) THEN 
      return_arr_.extend;
      IF (total_weight_ > 0) THEN 
         return_arr_(return_arr_.last).weight := round(total_weight_, 4);
      END IF;
      return_arr_(return_arr_.last).product_details_list := product_details_;
      
      Client_SYS.Clear_Attr(attr_);
      Client_SYS.Add_To_Attr('HANDLING_UNIT_ID', '', attr_);
      return_arr_(return_arr_.last).custom_fields := Get_Custom_Fields_Arr(freight_provider_, Frt_Message_Level_API.DB_HANDLING_UNIT, Frt_Source_Ref_Type_API.DB_SHIPMENT, to_char(shipment_id_), attr_);
      Get_Package_User_Fields(freight_provider_, return_arr_(return_arr_.last), Frt_Message_Level_API.DB_HANDLING_UNIT, Frt_Source_Ref_Type_API.DB_SHIPMENT, to_char(shipment_id_), attr_);
   END IF;
   RETURN return_arr_;                       
END Get_Package_Details_Arr;

FUNCTION Get_Shipment_Message_Arr (
   freight_provider_ IN VARCHAR2,
   source_ref_type_  IN VARCHAR2,
   source_ref_       IN VARCHAR2) RETURN Sp_Shipment_Message_Arr
IS   
   return_arr_    Sp_Shipment_Message_Arr := Sp_Shipment_Message_Arr();
   mappings_arr_  Frt_Provider_Mapping_API.Frt_Provider_Mappings_Arr := Frt_Provider_Mapping_API.Frt_Provider_Mappings_Arr();
   attr_          VARCHAR2(200);
   lu_name_       VARCHAR2(100);
   field_type_    VARCHAR2(20);
   tag_value_     VARCHAR2(2000);
BEGIN
   mappings_arr_ := Frt_Provider_Mapping_API.Get_Frt_Provider_Mappings_Arr(
                           freight_provider_, 
                           Frt_Message_Level_API.Decode(Frt_Message_Level_API.DB_SHIPMENT), 
                           Frt_Map_Tag_Type_API.DB_MESSAGE_LIST);
   IF (mappings_arr_.COUNT > 0) THEN 
      FOR i IN mappings_arr_.FIRST..mappings_arr_.LAST LOOP    
         lu_name_ := Frt_Int_Utility_API.Get_Lu_By_Source_Field_Type(mappings_arr_(i).source_field_type, field_type_);
         IF (lu_name_ IS NOT NULL AND source_ref_type_ = Frt_Source_Ref_Type_API.DB_SHIPMENT ) THEN 
            IF (lu_name_ = 'Shipment') THEN 
               Client_SYS.Add_To_Attr('SHIPMENT_ID', source_ref_, attr_);
            ELSIF (lu_name_ = 'CustomerOrder') THEN 
               Client_SYS.Add_To_Attr('ORDER_NO', Shipment_API.Get_Source_Ref1(source_ref_), attr_);
            ELSIF (lu_name_ = 'FrtShippingInfo') THEN
               Client_SYS.Add_To_Attr('SOURCE_REF', source_ref_, attr_);
               Client_SYS.Add_To_Attr('SOURCE_REF_TYPE', Frt_Source_Ref_Type_API.Decode(source_ref_type_), attr_);
            END IF;

            IF (field_type_ = 'CORE') THEN 
               tag_value_ := Frt_Provider_Mapping_API.Get_Field_Value(lu_name_, mappings_arr_(i).source_field_name, attr_);
            ELSIF (field_type_ = 'CUST') THEN 
               tag_value_ := Frt_Provider_Mapping_API.Get_Custom_Field_Value(lu_name_, mappings_arr_(i).source_field_name, attr_);
            END IF;
            IF (mappings_arr_(i).map_tag_name IS NOT NULL AND tag_value_ IS NOT NULL) THEN 
               return_arr_.extend;
               return_arr_(return_arr_.last).name := mappings_arr_(i).map_tag_name;
               return_arr_(return_arr_.last).value := tag_value_;
            END IF;
         END IF;
      END LOOP;
   END IF;
   RETURN return_arr_;                                        
END Get_Shipment_Message_Arr;

   part_no_       inventory_part.part_no%TYPE;
   catalog_no_    sales_part.catalog_no%TYPE;
   char_code_     inventory_part_char.characteristic_code%TYPE;   
   
   CURSOR get_sales_part_char(char_code_ IN VARCHAR2) IS
         select attr_value
         FROM  sales_part_characteristic_tab
         WHERE contract = contract_
         AND   catalog_no = catalog_no_
         AND   characteristic_code = char_code_;
   
   CURSOR get_key_cols IS
         SELECT column_name
         FROM   dictionary_sys_view_column
         WHERE  view_name = Dictionary_SYS.Get_Base_View(lu_name_)
         AND    type_flag IN ('P', 'K')
         ORDER BY column_index;
BEGIN
   mappings_arr_ := Frt_Provider_Mapping_API.Get_Frt_Provider_Mappings_Arr(
                              freight_provider_, 
                              Frt_Message_Level_API.Decode(message_level_db_), 
                              Frt_Map_Tag_Type_API.DB_CUSTOM_FIELD);
   IF (mappings_arr_.COUNT > 0) THEN 
      FOR i IN mappings_arr_.FIRST..mappings_arr_.LAST LOOP    
         lu_name_ := Frt_Int_Utility_API.Get_Lu_By_Source_Field_Type(mappings_arr_(i).source_field_type, field_type_);  
         Client_SYS.Clear_Attr(attr_);
         IF (lu_name_ IS NOT NULL) THEN

                     END IF;
                  END IF;
               ELSIF (message_level_db_ = Frt_Message_Level_API.DB_SHIPMENT AND source_ref_type_ = Frt_Source_Ref_Type_API.DB_CUSTOMER_ORDER) THEN 
                  IF (lu_name_ = 'CustomerOrder') THEN 
                     Client_SYS.Add_To_Attr('ORDER_NO', source_ref_, attr_);
                  END IF;
               END IF;
            END IF;
            
            IF (field_type_ = 'CORE') THEN 
               tag_value_ := Frt_Provider_Mapping_API.Get_Field_Value(lu_name_, mappings_arr_(i).source_field_name, attr_);
            ELSIF (field_type_ = 'CUST') THEN 
               tag_value_ := Frt_Provider_Mapping_API.Get_Custom_Field_Value(lu_name_, mappings_arr_(i).source_field_name, attr_);
            ELSIF (field_type_ = 'INVCHAR') THEN 
               char_code_ := mappings_arr_(i).source_field_name;
               tag_value_ := Inventory_Part_Char_API.Get_Attr_Value(contract_, part_no_, char_code_);
            ELSIF (field_type_ = 'SALECHAR') THEN 
               char_code_ := mappings_arr_(i).source_field_name;
               OPEN get_sales_part_char(char_code_);
               FETCH get_sales_part_char INTO tag_value_;
               CLOSE get_sales_part_char;
            END IF;
            
            IF (mappings_arr_(i).map_tag_name IS NOT NULL AND tag_value_ IS NOT NULL) THEN 
               return_arr_.extend;
               return_arr_(return_arr_.last).name := mappings_arr_(i).map_tag_name;
               IF (mappings_arr_(i).map_tag_name = 'ItemVolumeUnit') THEN 
                  tag_value_ := UPPER(tag_value_);
               END IF;
               return_arr_(return_arr_.last).value := tag_value_;
            END IF;
         END IF;
      END LOOP;
   END IF;
   RETURN return_arr_;                                                                      
END Get_Custom_Fields_Arr;

-- Copy of Get_Custom_Fields_Arr - Used to retrieve mappings for additional tags at SAME LEVEL
-- as JSON under construction. Can be used for tags without hardcoded mapping, or to override harcoded mapping.
FUNCTION Get_Map_Tags_Arr (
   freight_provider_ IN VARCHAR2,
   message_level_db_ IN VARCHAR2,
   map_tag_type_db_  IN VARCHAR2,
   source_ref_type_  IN VARCHAR2,
   source_ref_       IN VARCHAR2,
   key_attr_         IN VARCHAR2 DEFAULT NULL) RETURN Sp_Map_Tags_Arr
IS
   return_arr_    Sp_Map_Tags_Arr := Sp_Map_Tags_Arr();
   mappings_arr_  Frt_Provider_Mapping_API.Frt_Provider_Mappings_Arr := Frt_Provider_Mapping_API.Frt_Provider_Mappings_Arr();
   attr_          VARCHAR2(200);
   lu_name_       VARCHAR2(100);
   field_type_    VARCHAR2(20);
   tag_value_     VARCHAR2(2000);
   contract_      inventory_part.contract%TYPE;
   part_no_       inventory_part.part_no%TYPE;
   catalog_no_    sales_part.catalog_no%TYPE;
   char_code_     inventory_part_char.characteristic_code%TYPE;   
   
   CURSOR get_sales_part_char(char_code_ IN VARCHAR2) IS
         select attr_value
         FROM  sales_part_characteristic_tab
         WHERE contract = contract_
         AND   catalog_no = catalog_no_
         AND   characteristic_code = char_code_;
   
   CURSOR get_key_cols IS
         SELECT column_name
         FROM   dictionary_sys_view_column
         WHERE  view_name = Dictionary_SYS.Get_Base_View(lu_name_)
         AND    type_flag IN ('P', 'K')
         ORDER BY column_index;
BEGIN
   mappings_arr_ := Frt_Provider_Mapping_API.Get_Frt_Provider_Mappings_Arr(
                              freight_provider_, 
                              Frt_Message_Level_API.Decode(message_level_db_), 
                              map_tag_type_db_);
   IF (mappings_arr_.COUNT > 0) THEN 
      FOR i IN mappings_arr_.FIRST..mappings_arr_.LAST LOOP    
         lu_name_ := Frt_Int_Utility_API.Get_Lu_By_Source_Field_Type(mappings_arr_(i).source_field_type, field_type_);  
         Client_SYS.Clear_Attr(attr_);
         IF (lu_name_ IS NOT NULL) THEN 
            IF (source_ref_type_ = Frt_Source_Ref_Type_API.DB_SHIPMENT) THEN 
               IF (lu_name_ = 'Shipment' AND message_level_db_ = Frt_Message_Level_API.DB_SHIPMENT) THEN 
                  Client_SYS.Add_To_Attr('SHIPMENT_ID', source_ref_, attr_);
               ELSIF (lu_name_ = 'CustomerOrder'  AND message_level_db_ = Frt_Message_Level_API.DB_SHIPMENT) THEN 
                  Client_SYS.Add_To_Attr('ORDER_NO', Shipment_API.Get_Source_Ref1(source_ref_), attr_);
               ELSIF (lu_name_ = 'HandlingUnit'  AND message_level_db_ = Frt_Message_Level_API.DB_HANDLING_UNIT) THEN 
                  attr_ := key_attr_;            
               ELSIF (message_level_db_ = Frt_Message_Level_API.DB_LINE_ITEM) THEN 
                  IF (lu_name_ = 'CustomerOrderLine') THEN 
                     Client_SYS.Add_To_Attr('ORDER_NO', Client_SYS.Get_Item_Value('ORDER_NO', key_attr_), attr_);
                     Client_SYS.Add_To_Attr('LINE_NO', Client_SYS.Get_Item_Value('LINE_NO', key_attr_), attr_);
                     Client_SYS.Add_To_Attr('RELEASE_NO', Client_SYS.Get_Item_Value('REL_NO', key_attr_), attr_);
                     Client_SYS.Add_To_Attr('LINE_ITEM_NO', Client_SYS.Get_Item_Value('LINE_ITEM_NO', key_attr_), attr_);
                  ELSIF (lu_name_ = 'SalesPart') THEN 
                     Client_SYS.Add_To_Attr('CONTRACT', Client_SYS.Get_Item_Value('CONTRACT', key_attr_), attr_);
                     Client_SYS.Add_To_Attr('CATALOG_NO', Client_SYS.Get_Item_Value('CATALOG_NO', key_attr_), attr_);
                  ELSIF (lu_name_ = 'SalesPartCharacteristic') THEN 
                     contract_ := Client_SYS.Get_Item_Value('CONTRACT', key_attr_);
                     catalog_no_ := Client_SYS.Get_Item_Value('CATALOG_NO', key_attr_);
                  ELSIF (lu_name_ = 'InventoryPart') THEN 
                     Client_SYS.Add_To_Attr('CONTRACT', Client_SYS.Get_Item_Value('CONTRACT', key_attr_), attr_);
                     Client_SYS.Add_To_Attr('PART_NO', Client_SYS.Get_Item_Value('PART_NO', key_attr_), attr_);
                  ELSIF (lu_name_ = 'InventoryPartChar') THEN 
                     contract_ := Client_SYS.Get_Item_Value('CONTRACT', key_attr_);
                     part_no_ := Client_SYS.Get_Item_Value('PART_NO', key_attr_);
                  ELSIF (lu_name_ = 'PartCatalog') THEN 
                     Client_SYS.Add_To_Attr('PART_NO', Client_SYS.Get_Item_Value('PART_NO', key_attr_), attr_);
                  END IF;
               END IF;                        
            ELSE
               IF (message_level_db_ = Frt_Message_Level_API.DB_LINE_ITEM) THEN 
                  IF (lu_name_ = 'SalesPart') THEN 
                     Client_SYS.Add_To_Attr('CONTRACT', Client_SYS.Get_Item_Value('CONTRACT', key_attr_), attr_);
                     Client_SYS.Add_To_Attr('CATALOG_NO', Client_SYS.Get_Item_Value('CATALOG_NO', key_attr_), attr_);
                  ELSIF (lu_name_ = 'SalesPartCharacteristic') THEN 
                     contract_ := Client_SYS.Get_Item_Value('CONTRACT', key_attr_);
                     catalog_no_ := Client_SYS.Get_Item_Value('CATALOG_NO', key_attr_);
                  ELSIF (lu_name_ = 'InventoryPart') THEN 
                     Client_SYS.Add_To_Attr('CONTRACT', Client_SYS.Get_Item_Value('CONTRACT', key_attr_), attr_);
                     Client_SYS.Add_To_Attr('PART_NO', Client_SYS.Get_Item_Value('PART_NO', key_attr_), attr_);
                  ELSIF (lu_name_ = 'InventoryPartChar') THEN 
                     contract_ := Client_SYS.Get_Item_Value('CONTRACT', key_attr_);
                     part_no_ := Client_SYS.Get_Item_Value('PART_NO', key_attr_);
                  ELSIF (lu_name_ = 'PartCatalog') THEN 
                     Client_SYS.Add_To_Attr('PART_NO', Client_SYS.Get_Item_Value('PART_NO', key_attr_), attr_);
                  END IF;
                  
                  IF (source_ref_type_ = Frt_Source_Ref_Type_API.DB_CUSTOMER_ORDER) THEN
                     IF (lu_name_ = 'CustomerOrderLine') THEN 
                        Client_SYS.Add_To_Attr('ORDER_NO', Client_SYS.Get_Item_Value('ORDER_NO', key_attr_), attr_);
                        Client_SYS.Add_To_Attr('LINE_NO', Client_SYS.Get_Item_Value('LINE_NO', key_attr_), attr_);
                        Client_SYS.Add_To_Attr('RELEASE_NO', Client_SYS.Get_Item_Value('REL_NO', key_attr_), attr_);
                        Client_SYS.Add_To_Attr('LINE_ITEM_NO', Client_SYS.Get_Item_Value('LINE_ITEM_NO', key_attr_), attr_);
                     END IF;
                  END IF;
               ELSIF (message_level_db_ = Frt_Message_Level_API.DB_SHIPMENT AND source_ref_type_ = Frt_Source_Ref_Type_API.DB_CUSTOMER_ORDER) THEN 
                  IF (lu_name_ = 'CustomerOrder') THEN 
                     Client_SYS.Add_To_Attr('ORDER_NO', source_ref_, attr_);
                  END IF;
               END IF;
            END IF;
            
            IF (field_type_ = 'CORE') THEN 
               tag_value_ := Frt_Provider_Mapping_API.Get_Field_Value(lu_name_, mappings_arr_(i).source_field_name, attr_);
            ELSIF (field_type_ = 'CUST') THEN 
               tag_value_ := Frt_Provider_Mapping_API.Get_Custom_Field_Value(lu_name_, mappings_arr_(i).source_field_name, attr_);
            ELSIF (field_type_ = 'INVCHAR') THEN 
               char_code_ := mappings_arr_(i).source_field_name;
               tag_value_ := Inventory_Part_Char_API.Get_Attr_Value(contract_, part_no_, char_code_);
            ELSIF (field_type_ = 'SALECHAR') THEN 
               char_code_ := mappings_arr_(i).source_field_name;
               OPEN get_sales_part_char(char_code_);
               FETCH get_sales_part_char INTO tag_value_;
               CLOSE get_sales_part_char;
            END IF;

            IF (mappings_arr_(i).map_tag_name IS NOT NULL AND tag_value_ IS NOT NULL) THEN 
               return_arr_.extend;
               return_arr_(return_arr_.last).name := mappings_arr_(i).map_tag_name;
               return_arr_(return_arr_.last).value := tag_value_;
            END IF;
         END IF;
      END LOOP;
   END IF;
   RETURN return_arr_;                                                                      
END Get_Map_Tags_Arr;
