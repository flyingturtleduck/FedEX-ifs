
CREATE OR REPLACE PROCEDURE parse_rate_response (
   p_json   IN  CLOB
)
IS
   
   PROCEDURE fld(p_label VARCHAR2, p_value VARCHAR2) IS
   BEGIN
      DBMS_OUTPUT.PUT_LINE('  ' || RPAD(p_label, 26) || NVL(p_value, '(none)'));
   END fld;

   PROCEDURE put_row(
      p_carrier VARCHAR2, p_name VARCHAR2, p_svc_code VARCHAR2,
      p_ship_via VARCHAR2, p_rate_type VARCHAR2,
      p_cost NUMBER, p_price NUMBER, p_list NUMBER,
      p_curr VARCHAR2, p_arr_date VARCHAR2
   ) IS
   BEGIN
      DBMS_OUTPUT.PUT_LINE(
         RPAD(p_carrier,8) || '  ' || RPAD(p_name,32) || '  ' ||
         RPAD(NVL(p_svc_code,'?'),8) || '  ' || RPAD(p_ship_via,30) || '  ' ||
         RPAD(p_rate_type,20) || '  ' ||
         LPAD(TO_CHAR(p_cost, 'FM999990.00'),12)  || '  ' ||
         LPAD(TO_CHAR(p_price,'FM999990.00'),12)  || '  ' ||
         LPAD(TO_CHAR(p_list, 'FM999990.00'),12)  || '  ' ||
         RPAD(p_curr,4) || '  ' || NVL(p_arr_date,'N/A')
      );
   END put_row;

   PROCEDURE put_detail(
      p_carrier VARCHAR2, p_name VARCHAR2, p_svc_code VARCHAR2,
      p_ship_via VARCHAR2, p_cost NUMBER, p_price NUMBER,
      p_list NUMBER, p_curr VARCHAR2, p_arr_date VARCHAR2
   ) IS
   BEGIN
      fld('Carrier',              p_carrier);
      fld('Carrier Name',         p_name);
      fld('Carrier Service Code', NVL(p_svc_code,'N/A'));
      fld('Ship Via',             p_ship_via);
      fld('Total Cost',           TO_CHAR(p_cost, 'FM999990.00') || ' ' || p_curr);
      fld('Total Price',          TO_CHAR(p_price,'FM999990.00') || ' ' || p_curr);
      fld('Total List Price',     TO_CHAR(p_list, 'FM999990.00') || ' ' || p_curr);
      fld('Currency Code',        p_curr);
      fld('Arrival Date',         NVL(p_arr_date,'N/A'));
   END put_detail;

BEGIN
   DBMS_OUTPUT.ENABLE(1000000);


   FOR r IN (
      WITH rate_raw AS (
         
         SELECT
            jt.service_type,
            jt.service_name,
            jt.rate_type,
            jt.total_base_charge,
            jt.total_net_charge,
            jt.currency,
            COALESCE(
               jt.svc_desc_code,
               jt.op_svc_code,
               MAX(jt.nested_svc_code) OVER (PARTITION BY jt.service_type)
            ) AS carrier_svc_code,
            COALESCE(
               jt.commit_date,
               MAX(jt.nested_commit_dt) OVER (PARTITION BY jt.service_type),
               MAX(jt.nested_op_commit) OVER (PARTITION BY jt.service_type)
            ) AS arrival_date
         FROM JSON_TABLE(
            p_json,
            '$.output.rateReplyDetails[*]'
            COLUMNS (
               service_type  VARCHAR2(100) PATH '$.serviceType',
               service_name  VARCHAR2(200) PATH '$.serviceName',
               op_svc_code   VARCHAR2(20)  PATH '$.operationalDetail.serviceCode',
               svc_desc_code VARCHAR2(20)  PATH '$.serviceDescription.code',
               commit_date   VARCHAR2(50)  PATH '$.commit.dateDetail.dayCxsFormat',
               NESTED PATH '$.ratedShipmentDetails[*]' COLUMNS (
                  rate_type         VARCHAR2(50) PATH '$.rateType',
                  total_base_charge NUMBER       PATH '$.totalBaseCharge',
                  total_net_charge  NUMBER       PATH '$.totalNetCharge',
                  currency          VARCHAR2(10) PATH '$.currency',
                  nested_svc_code   VARCHAR2(20) PATH '$.serviceDescription.code',
                  nested_commit_dt  VARCHAR2(50) PATH '$.commit.dateDetail.dayCxsFormat',
                  nested_op_commit  VARCHAR2(50) PATH '$.operationalDetail.commitDate'
               )
            )
         ) jt
      ),
      list_px AS (
         -- List price per service — shown on every rate row for that service
         SELECT service_type, total_net_charge AS list_price
         FROM   rate_raw
         WHERE  rate_type = 'LIST'
      )
      SELECT
         'DEMO'                AS carrier,
         r.service_name,
         r.carrier_svc_code,
         r.service_type         AS ship_via,
         r.rate_type,
         r.total_base_charge,
         r.total_net_charge,
         NVL(lp.list_price, 0) AS list_price,
         r.currency,
         r.arrival_date
      FROM      rate_raw     r
      LEFT JOIN list_px lp ON lp.service_type = r.service_type
      ORDER BY
         r.service_type,
         DECODE(r.rate_type,
            'ACCOUNT',             1,
            'LIST',                2,
            'PREFERRED_INCENTIVE', 3,
            'PREFERRED_CURRENCY',  4, 9)
   ) LOOP
      put_row(r.carrier, r.service_name, r.carrier_svc_code, r.ship_via,
              r.rate_type, r.total_base_charge, r.total_net_charge,
              r.list_price, r.currency, r.arrival_date);
   END LOOP;

   FOR r IN (
      WITH rate_raw AS (
         SELECT
            jt.service_type, jt.service_name, jt.rate_type,
            jt.total_base_charge, jt.total_net_charge, jt.currency,
            COALESCE(jt.svc_desc_code, jt.op_svc_code,
                     MAX(jt.nested_svc_code) OVER (PARTITION BY jt.service_type)) AS carrier_svc_code,
            COALESCE(jt.commit_date,
                     MAX(jt.nested_commit_dt) OVER (PARTITION BY jt.service_type),
                     MAX(jt.nested_op_commit) OVER (PARTITION BY jt.service_type)) AS arrival_date
         FROM JSON_TABLE(
            p_json, '$.output.rateReplyDetails[*]'
            COLUMNS (
               service_type  VARCHAR2(100) PATH '$.serviceType',
               service_name  VARCHAR2(200) PATH '$.serviceName',
               op_svc_code   VARCHAR2(20)  PATH '$.operationalDetail.serviceCode',
               svc_desc_code VARCHAR2(20)  PATH '$.serviceDescription.code',
               commit_date   VARCHAR2(50)  PATH '$.commit.dateDetail.dayCxsFormat',
               NESTED PATH '$.ratedShipmentDetails[*]' COLUMNS (
                  rate_type         VARCHAR2(50) PATH '$.rateType',
                  total_base_charge NUMBER       PATH '$.totalBaseCharge',
                  total_net_charge  NUMBER       PATH '$.totalNetCharge',
                  currency          VARCHAR2(10) PATH '$.currency',
                  nested_svc_code   VARCHAR2(20) PATH '$.serviceDescription.code',
                  nested_commit_dt  VARCHAR2(50) PATH '$.commit.dateDetail.dayCxsFormat',
                  nested_op_commit  VARCHAR2(50) PATH '$.operationalDetail.commitDate'
               )
            )
         ) jt
      ),
      list_px AS (
         SELECT service_type, total_net_charge AS list_price
         FROM   rate_raw WHERE rate_type = 'LIST'
      ),
      ranked AS (
         SELECT 'DEMO' AS carrier, r.service_name, r.carrier_svc_code,
                r.service_type, r.total_base_charge, r.total_net_charge,
                NVL(lp.list_price,0) AS list_price, r.currency, r.arrival_date,
                RANK() OVER (ORDER BY r.total_net_charge ASC) AS rnk
         FROM      rate_raw r
         LEFT JOIN list_px lp ON lp.service_type = r.service_type
         WHERE     r.rate_type = 'ACCOUNT'
      )
      SELECT * FROM ranked WHERE rnk = 1
   ) LOOP
      put_detail(r.carrier, r.service_name, r.carrier_svc_code, r.service_type,
                 r.total_base_charge, r.total_net_charge, r.list_price,
                 r.currency, r.arrival_date);
   END LOOP;

   FOR r IN (
      WITH rate_raw AS (
         SELECT
            jt.service_type, jt.service_name, jt.rate_type,
            jt.total_base_charge, jt.total_net_charge, jt.currency,
            COALESCE(jt.svc_desc_code, jt.op_svc_code,
                     MAX(jt.nested_svc_code) OVER (PARTITION BY jt.service_type)) AS carrier_svc_code,
            COALESCE(jt.commit_date,
                     MAX(jt.nested_commit_dt) OVER (PARTITION BY jt.service_type),
                     MAX(jt.nested_op_commit) OVER (PARTITION BY jt.service_type)) AS arrival_date
         FROM JSON_TABLE(
            p_json, '$.output.rateReplyDetails[*]'
            COLUMNS (
               service_type  VARCHAR2(100) PATH '$.serviceType',
               service_name  VARCHAR2(200) PATH '$.serviceName',
               op_svc_code   VARCHAR2(20)  PATH '$.operationalDetail.serviceCode',
               svc_desc_code VARCHAR2(20)  PATH '$.serviceDescription.code',
               commit_date   VARCHAR2(50)  PATH '$.commit.dateDetail.dayCxsFormat',
               NESTED PATH '$.ratedShipmentDetails[*]' COLUMNS (
                  rate_type         VARCHAR2(50) PATH '$.rateType',
                  total_base_charge NUMBER       PATH '$.totalBaseCharge',
                  total_net_charge  NUMBER       PATH '$.totalNetCharge',
                  currency          VARCHAR2(10) PATH '$.currency',
                  nested_svc_code   VARCHAR2(20) PATH '$.serviceDescription.code',
                  nested_commit_dt  VARCHAR2(50) PATH '$.commit.dateDetail.dayCxsFormat',
                  nested_op_commit  VARCHAR2(50) PATH '$.operationalDetail.commitDate'
               )
            )
         ) jt
      ),
      list_px AS (
         SELECT service_type, total_net_charge AS list_price
         FROM   rate_raw WHERE rate_type = 'LIST'
      ),
      ranked AS (
         SELECT 'DEMO' AS carrier, r.service_name, r.carrier_svc_code,
                r.service_type, r.total_base_charge, r.total_net_charge,
                NVL(lp.list_price,0) AS list_price, r.currency, r.arrival_date,
                RANK() OVER (
                   ORDER BY
                      CASE WHEN r.arrival_date IS NOT NULL
                           THEN TO_DATE(SUBSTR(r.arrival_date,1,19),
                                        'YYYY-MM-DD"T"HH24:MI:SS')
                           ELSE DATE '9999-12-31'
                      END ASC
                ) AS rnk
         FROM      rate_raw r
         LEFT JOIN list_px lp ON lp.service_type = r.service_type
         WHERE     r.rate_type = 'ACCOUNT'
      )
      SELECT * FROM ranked WHERE rnk = 1
   ) LOOP
      put_detail(r.carrier, r.service_name, r.carrier_svc_code, r.service_type,
                 r.total_base_charge, r.total_net_charge, r.list_price,
                 r.currency, r.arrival_date);
   END LOOP;

EXCEPTION
   WHEN OTHERS THEN
      DBMS_OUTPUT.PUT_LINE('ERROR: ' || SQLERRM);
      RAISE;
END parse_rate_response;
