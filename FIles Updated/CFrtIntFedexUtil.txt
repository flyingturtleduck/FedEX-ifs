PROCEDURE Get_Frt_Rate_License_Values___(
   freight_provider_ IN  VARCHAR2,
   rate_client_id_ OUT VARCHAR2,
   rate_client_secret_ OUT VARCHAR2)
IS  
   CURSOR get_config_values IS
         SELECT upper(config_id) config_id, config_value 
         FROM frt_provider_config_value_tab
         WHERE freight_provider_id = freight_provider_;
BEGIN
   FOR rec_ IN get_config_values LOOP
      IF (rec_.config_id = 'CLIENT_ID') THEN 
         rate_client_id_ := rec_.config_value;
      ELSIF (rec_.config_id = 'CLIENT_SECRET') THEN 
         rate_client_secret_ := rec_.config_value;
      END IF;
   END LOOP;
END Get_Frt_Rate_License_Values___;

FUNCTION Get_Freight_Location___(
   freight_provider_ IN VARCHAR2) RETURN VARCHAR2
IS  
   frt_provider_       frt_provider_config_value_tab.freight_provider_id%TYPE:=freight_provider_;
   api_location_       frt_provider_config_value_tab.config_value%TYPE;
   
   CURSOR get_config_values IS
         SELECT upper(config_id) config_id, config_value 
         FROM frt_provider_config_value_tab
         WHERE freight_provider_id = frt_provider_;
BEGIN
   IF(frt_provider_ IS NULL) THEN 
      frt_provider_ := Frt_Int_Ship_Util_API.Get_Freight_Provider;
   END IF;
   
   FOR rec_ IN get_config_values LOOP
      IF (rec_.config_id = 'FEDEX LOCATION') THEN 
         api_location_ := rec_.config_value;
         EXIT;
      END IF;
   END LOOP;
   
   RETURN api_location_;  
END Get_Freight_Location___;

FUNCTION Get_Bearer_Token(
   freight_provider_ IN VARCHAR2
   ) RETURN VARCHAR2
IS
   rate_client_id_         frt_provider_config_value_tab.config_value%TYPE;
   rate_client_secret_     frt_provider_config_value_tab.config_value%TYPE;
   request_headers_        VARCHAR2(500);
   json_clob_              CLOB;
   token_                  VARCHAR2(2000);
   query_params_           Plsqlap_Document_API.Document;
BEGIN
   Get_Frt_Rate_License_Values___(freight_provider_, rate_client_id_, rate_client_secret_);
   request_headers_ := Get_Auth_Request_Header___(freight_provider_);
   json_clob_ := 'grant_type=client_credentials&client_id='||rate_client_id_||'&client_secret='||rate_client_secret_;
   
   
   Plsql_Rest_Sender_API.Call_Rest_EndPoint_Json_Sync(
               rest_service_ => 'FEDEX_AUTH_SERVICE', 
               json_ => json_clob_, 
               http_method_ => 'POST',
               http_req_headers_ => request_headers_,
               query_parameters_ => query_params_,
               sender_ => Fnd_Session_API.Get_Fnd_User,
               receiver_ => freight_provider_);
   
   -- dbms_output.Put_Line(json_clob_); 
   IF (json_clob_ IS NOT NULL) THEN 
     token_ := JSON_VALUE(json_clob_ , '$.access_token');
     RETURN token_;
   END IF; 
   
   RETURN to_char(NULL);
END;

-- Method to post to FexEx Rate API
FUNCTION Get_Freight_Rates(
   freight_provider_ IN VARCHAR2,
   freight_shipment_ IN Json_Object_T) RETURN Json_Object_T
IS
   
   query_params_           Plsqlap_Document_API.Document;
   url_params_             Plsqlap_Document_API.Document;
   request_headers_        VARCHAR2(500);
   json_clob_              CLOB;
   shipment_rates_obj_     Json_Object_T := Json_Object_T();
BEGIN  
   
   Get_Rest_Parameters___(query_params_, url_params_, request_headers_, freight_provider_);
   -- build json payload
   IF (freight_shipment_ IS NOT NULL) THEN 
      json_clob_ := freight_shipment_.to_clob();
   END IF;  
   -- dbms_output.Put_Line(json_clob_); 
   
   Plsql_Rest_Sender_API.Call_Rest_EndPoint_Json_Sync(
               rest_service_ => 'RECEIVE_SHIPMENT_RATES_FROM_FEDEX', 
               json_ => json_clob_, 
               http_method_ => 'POST',
               http_req_headers_ => request_headers_,
               query_parameters_ => query_params_,
               sender_ => Fnd_Session_API.Get_Fnd_User,
               receiver_ => freight_provider_);
   
   -- dbms_output.Put_Line(json_clob_); 
   IF (json_clob_ IS NOT NULL) THEN 
      shipment_rates_obj_ := json_object_t.parse(json_clob_);
   END IF;     
   
   RETURN shipment_rates_obj_;  
END Get_Freight_Rates;

PROCEDURE Get_Rest_Parameters___(
   query_params_     OUT Plsqlap_Document_API.Document,
   url_params_       OUT Plsqlap_Document_API.Document, 
   request_headers_  OUT VARCHAR2,
   freight_provider_ IN  VARCHAR2,
   url_parameter1_   IN  VARCHAR2 DEFAULT NULL)
IS
   api_location_           frt_provider_config_value_tab.config_value%TYPE;
   api_license_key_        frt_provider_config_value_tab.config_value%TYPE; 
   rate_license_key_       frt_provider_config_value_tab.config_value%TYPE;
   rate_ups_lic_key_       frt_provider_config_value_tab.config_value%TYPE;
   api_ver_                frt_provider_config_value_tab.config_value%TYPE;
BEGIN
   Get_Freight_Config_Values___(freight_provider_, api_location_, api_license_key_, rate_license_key_, rate_ups_lic_key_, api_ver_);
   request_headers_ := Get_Request_Headers___(freight_provider_, api_location_, api_license_key_);
   
   IF (freight_provider_ = 'PACEJET') THEN  
      Error_SYS.Check_Not_Null(lu_name_, 'API_LICENSE_KEY', api_license_key_);
      Error_SYS.Check_Not_Null(lu_name_, 'API_LOCATION', api_location_);
   END IF;
   
   IF (url_parameter1_ IS NOT NULL) THEN    
      url_params_ := Plsqlap_Document_API.New_Document('url_params');
      Plsqlap_Document_API.Add_Attribute(url_params_,'parameter1', url_parameter1_);
   END IF;
   
   query_params_ := Plsqlap_Document_API.New_Document('QUERY_PARAMETERS');
   IF(api_ver_ IS NOT NULL) THEN 
      Plsqlap_Document_API.Add_Attribute(query_params_,'api-version', api_ver_);
   END IF;
END Get_Rest_Parameters___;

PROCEDURE Get_Freight_Config_Values___(
   freight_provider_ IN  VARCHAR2,
   api_location_     OUT VARCHAR2,
   api_license_key_  OUT VARCHAR2,
   rate_license_key_ OUT VARCHAR2,
   rate_ups_lic_key_ OUT VARCHAR2,
   api_ver_          OUT VARCHAR2)
IS   
   CURSOR get_config_values IS
         SELECT upper(config_id) config_id, config_value 
         FROM frt_provider_config_value_tab
         WHERE freight_provider_id = freight_provider_;
BEGIN
   FOR rec_ IN get_config_values LOOP
      IF (rec_.config_id = 'API LICENSE KEY') THEN 
         api_license_key_ := rec_.config_value;
      ELSIF (rec_.config_id = 'API VERSION') THEN 
         api_ver_ := rec_.config_value;
      ELSIF (rec_.config_id = 'PACEJET LOCATION') THEN 
         api_location_ := rec_.config_value;
      ELSIF (rec_.config_id = 'API RATES LICENSE ID') THEN 
         rate_license_key_ := rec_.config_value;
      ELSIF (rec_.config_id = 'API RATES UPS LICENSE ID') THEN 
         rate_ups_lic_key_ := rec_.config_value;
      END IF;
   END LOOP;
END Get_Freight_Config_Values___;

FUNCTION Get_Auth_Request_Header___
   (freight_provider_ IN VARCHAR2)RETURN VARCHAR2
IS
   BEGIN 
   IF (freight_provider_ LIKE 'FEDEX%') THEN 
      RETURN 'Content-Type: application/x-www-form-urlencoded';
   ELSE
      RETURN NULL;
   END IF; 
END Get_Auth_Request_Header___;

FUNCTION Get_Request_Headers___(
   freight_provider_ IN VARCHAR2,
   api_location_     IN VARCHAR2 DEFAULT NULL,
   api_license_key_  IN VARCHAR2 DEFAULT NULL) RETURN VARCHAR2
IS  
BEGIN 
   IF (freight_provider_ = 'PACEJET') THEN 
      RETURN 'PacejetLocation:' || api_location_ || ', PacejetLicenseKey:' ||api_license_key_;
   ELSE
      RETURN NULL;
   END IF;  
END Get_Request_Headers___;

PROCEDURE Get_Cust_Order_Rates(
   shipment_rates_ OUT Frt_Int_Message_Util_API.Sr_Shipment_Rates_Rec,
   transaction_id_ IN  VARCHAR2,
   order_no_       IN  VARCHAR2,
   error_message_  OUT VARCHAR2,
   origin_         IN  Frt_Int_Message_Util_API.Sp_Origin_Rec,
   auto_pack_      IN  VARCHAR2,
   ship_via_code_  IN  VARCHAR2,
   frt_provider_   IN  VARCHAR2 DEFAULT NULL)
IS
   
   rate_client_id_         frt_provider_config_value_tab.config_value%TYPE;
   rate_client_secret_     frt_provider_config_value_tab.config_value%TYPE;
   freight_provider_       frt_provider_config_value_tab.freight_provider_id%TYPE:=frt_provider_;
   api_location_           frt_provider_config_value_tab.config_value%TYPE;
   shipment_rec_           Frt_Int_Message_Util_API.Sp_Shipment_Rec;
   freight_shipment_       Json_Object_T := Json_Object_T();
   shipment_rates_obj_     Json_Object_T := Json_Object_T();
   rate_obj_               Json_Object_T := Json_Object_T();
   j_obj_                  Json_Object_T := Json_Object_T();
   result_                 VARCHAR2(500);
   token_                  VARCHAR2(2000);
BEGIN   
   IF (freight_provider_ IS NULL) THEN 
      freight_provider_ := Frt_Int_Ship_Util_API.Get_Freight_Provider;
   END IF;
   Get_Frt_Rate_License_Values___(freight_provider_, rate_client_id_, rate_client_secret_);
   --Oauth2.0 token (bearer)
   token_ := Get_Bearer_Token(freight_provider_);
  -- prepare the pauyload json before this ins a separate procedure and use order_no to build
   --call the interface method Get_Freight_Rates to return the output json as clob
   shipment_rates_obj_ := Get_Freight_Rates(freight_provider_, freight_shipment_);
   
   IF (shipment_rates_obj_ IS NOT NULL) THEN 
      rate_obj_ := shipment_rates_obj_.get_object('rateRequestControlParameters');   
   END IF;  
   
   IF (rate_obj_ IS NULL) THEN 
      Frt_Int_Ship_Util_API.Process_Response_Error( error_message_,
                              freight_provider_,
                              freight_shipment_.get_string('TransactionId'),
                              'RECEIVE_ORDER_RATES_FROM_FEDEX',
                              Frt_Source_Ref_Type_API.DB_CUSTOMER_ORDER,
                              order_no_,
                              Frt_Message_Type_API.DB_RATE,
                              freight_shipment_,
                              shipment_rates_obj_); 
   ELSE  
      shipment_rates_ := Frt_Int_Message_Util_API.Json_To_Shipment_Rates_Rec(transaction_id_, freight_provider_, shipment_rates_obj_);  
      
      Frt_Int_Ship_Util_API.Log_Shipment_Trace(freight_provider_,
                                               transaction_id_,
                                               'RECEIVE_ORDER_RATES_FROM_FEDEX',
                                               Frt_Source_Ref_Type_API.DB_CUSTOMER_ORDER,
                                               order_no_,
                                               Frt_Message_Type_API.DB_RATE,
                                               freight_shipment_,
                                               shipment_rates_obj_,
                                               'Success');
   END IF;
EXCEPTION
   WHEN OTHERS THEN
      error_message_ := substr(sqlerrm,1, 2000);   
      -- update error log table  
      Frt_Int_Ship_Util_API.Log_Shipment_Trace(freight_provider_,
                                               transaction_id_,
                                               'RECEIVE_ORDER_RATES_FROM_FEDEX',
                                               Frt_Source_Ref_Type_API.DB_CUSTOMER_ORDER,
                                               order_no_,
                                               Frt_Message_Type_API.DB_RATE,
                                               freight_shipment_,
                                               shipment_rates_obj_,
                                               'Warning',
                                               error_message_);                                
      
END Get_Cust_Order_Rates;
